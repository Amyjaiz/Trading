let _currentModel = 'hm';
let _filterStocks = {hm: [], ts: [], nse: []};
let _allResults   = {hm: [], ts: [], nse: []};
let _universe     = null;

const UNI_KEYS = {
  hm:  {bot: 'hm',            type: 'add_remove'},
  ts:  {bot: 'triple_screen', type: 'add_remove'},
  nse: {bot: 'nse200',        type: 'extra'},
};

function switchModel(m) {
  _currentModel = m;
  document.querySelectorAll('.bt-tab').forEach(t => t.classList.toggle('active', t.dataset.model === m));
  document.querySelectorAll('.bt-panel').forEach(p => p.classList.toggle('active', p.id === 'panel-' + m));
}

function addFilterStock(e, model) {
  if (e.key !== 'Enter') return;
  const input = document.getElementById(model + '-stock-input');
  const sym = input.value.trim().toUpperCase();
  if (!sym) return;
  if (!_filterStocks[model].includes(sym)) {
    _filterStocks[model].push(sym);
    renderFilterTags(model);
  }
  input.value = '';
}

function removeFilterStock(model, sym) {
  _filterStocks[model] = _filterStocks[model].filter(s => s !== sym);
  renderFilterTags(model);
}

function renderFilterTags(model) {
  const wrap = document.getElementById(model + '-stock-tags');
  wrap.innerHTML = _filterStocks[model].map(s =>
    `${s}`
  ).join('');
}

async function loadUniverse() {
  _universe = await fetch('/api/universe').then(r => r.json());
  ['hm', 'ts', 'nse'].forEach(renderUniverse);
}

function renderUniverse(model) {
  if (!_universe) return;
  const {bot, type} = UNI_KEYS[model];
  const section = _universe[bot] || {};
  if (type === 'add_remove') {
    renderChips(model + '-uni-add',    section.add    || [], model, 'add');
    renderChips(model + '-uni-remove', section.remove || [], model, 'remove');
  } else {
    renderChips(model + '-uni-extra', section.extra || [], model, 'extra');
  }
}

function renderChips(elId, items, model, listKey) {
  const el = document.getElementById(elId);
  if (!el) return;
  el.innerHTML = items.length
    ? items.map(s =>
        `${s}`
      ).join('')
    : 'None';
}

function addUniverseStockBtn(model, listKey) {
  const input = document.getElementById(model + '-uni-' + listKey + '-input');
  if (!input) return;
  const sym = input.value.trim().toUpperCase();
  if (!sym) return;
  _addToUniverse(model, listKey, sym);
  input.value = '';
}

function addUniverseStock(e, model, listKey) {
  if (e.key !== 'Enter') return;
  addUniverseStockBtn(model, listKey);
}

async function _addToUniverse(model, listKey, sym) {
  if (!_universe) await loadUniverse();
  const {bot} = UNI_KEYS[model];
  _universe[bot] = _universe[bot] || {};
  const arr = _universe[bot][listKey] || [];
  if (!arr.includes(sym)) {
    arr.push(sym);
    _universe[bot][listKey] = arr;
    await _saveUniverse();
    renderUniverse(model);
    toast(`${sym} added to ${bot} ${listKey}`, 'success');
  }
}

async function removeUniverseStock(model, listKey, sym) {
  if (!_universe) return;
  const {bot} = UNI_KEYS[model];
  _universe[bot] = _universe[bot] || {};
  _universe[bot][listKey] = (_universe[bot][listKey] || []).filter(s => s !== sym);
  await _saveUniverse();
  renderUniverse(model);
  toast(`${sym} removed from ${bot} ${listKey}`, 'warn');
}

async function _saveUniverse() {
  await fetch('/api/universe', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(_universe),
  });
}

async function runBacktest(model, sweep=false) {
  const terminal = document.getElementById('bt-terminal-' + model);
  const progress = document.getElementById('progress-' + model);
  terminal.classList.remove('hidden');
  terminal.textContent = `Starting ${model.toUpperCase()} backtest...\n`;
  progress.style.display = 'block';
  document.getElementById('results-' + model).style.display = 'none';
  document.getElementById(model + '-run').disabled = true;
  document.getElementById(model + '-cancel').style.display = '';
  setBadge(model, 'running');

  const flags = [];
  if (sweep) flags.push('--sweep');
  if (model === 'hm') {
    const exit = document.getElementById('hm-exit').value;
    const ml   = document.getElementById('hm-ml').value;
    if (exit) flags.push(exit);
    if (ml)   flags.push(ml);
  }
  if (model === 'ts') {
    const trig = document.getElementById('ts-trigger').value;
    if (trig) flags.push(trig);
  }
  if (model === 'nse') {
    const slots = document.getElementById('nse-slots').value;
    const win   = document.getElementById('nse-window').value;
    const uni   = document.getElementById('nse-universe').value;
    if (slots) flags.push('--slots', slots);
    if (win)   flags.push(win);
    if (uni)   flags.push(uni);
  }

  const body = {
    model,
    stocks:  _filterStocks[model],
    start:   document.getElementById(model + '-start').value,
    capital: model !== 'nse' ? document.getElementById(model + '-capital').value : '',
    flags,
  };

  const res  = await fetch('/api/backtest/run', {
    method: 'POST', headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(body),
  });
  const data = await res.json();
  if (data.status !== 'started') {
    terminal.textContent += '\n' + (data.message || JSON.stringify(data));
    document.getElementById(model + '-run').disabled = false;
    document.getElementById(model + '-cancel').style.display = 'none';
    setBadge(model, 'idle');
    if (data.status === 'warning' || data.status === 'running') toast(data.message, 'warn');
  }
}

async function cancelBacktest(model) {
  await fetch('/api/backtest/cancel', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({model}),
  });
  setBadge(model, 'idle');
  document.getElementById(model + '-run').disabled = false;
  document.getElementById(model + '-cancel').style.display = 'none';
  document.getElementById('progress-' + model).style.display = 'none';
  toast(`${model.toUpperCase()} backtest cancelled`, 'warn');
}

function setBadge(model, status) {
  const badge = document.getElementById('badge-' + model);
  if (!badge) return;
  badge.textContent = status;
  badge.className = 'bt-status-badge status-' + status;
}

socket.on('backtest_output', d => {
  const terminal = document.getElementById('bt-terminal-' + d.model);
  if (!terminal) return;
  terminal.textContent += d.line + '\n';
  terminal.scrollTop = terminal.scrollHeight;
});

socket.on('backtest_done', d => {
  const m = d.model;
  document.getElementById(m + '-run').disabled = false;
  document.getElementById(m + '-cancel').style.display = 'none';
  document.getElementById('progress-' + m).style.display = 'none';
  const terminal = document.getElementById('bt-terminal-' + m);
  terminal.textContent += `\n-- Completed (exit ${d.returncode}) --\n`;
  terminal.scrollTop = terminal.scrollHeight;
  if (d.returncode === 0) {
    setBadge(m, 'completed');
    toast(`${m.toUpperCase()} backtest complete — loading results...`, 'success');
    loadResults(m);
  } else {
    setBadge(m, 'failed');
    toast(`${m.toUpperCase()} backtest exited with errors`, 'error');
  }
});

async function loadResults(model) {
  const res  = await fetch('/api/backtest/results/' + model);
  const data = await res.json();
  _allResults[model] = data.per_stock || [];
  const a = data.analytics || {};
  document.getElementById('bt-metrics-' + model).innerHTML = [
    ['CAGR',          a.cagr          ? a.cagr + '%'      : '—'],
    ['Sharpe',        a.sharpe        || '—'],
    ['Max DD',        a.maxdd         || '—'],
    ['Trades',        a.total_trades  || 0],
    ['Win Rate',      a.win_rate      ? a.win_rate + '%'  : '—'],
    ['Profit Factor', a.profit_factor || '—'],
    ['Avg Hold',      a.avg_hold      ? a.avg_hold + 'd' : '—'],
  ].map(([l, v]) =>
    `
${l}
${v}
`
  ).join('');
  document.getElementById('results-' + model).style.display = '';
  filterResults(model);
}

function filterResults(model) {
  const minWR  = parseFloat(document.getElementById(model + '-min-wr').value)     || 0;
  const minTrd = parseInt(document.getElementById(model + '-min-trades').value)   || 0;
  const sym    = (document.getElementById(model + '-sym-search').value || '').toUpperCase();
  const key    = document.getElementById(model + '-sort').value;
  let rows = (_allResults[model] || []).filter(r =>
    r.win_rate >= minWR && r.trades >= minTrd && (!sym || r.symbol.includes(sym))
  );
  rows.sort((a, b) => (b[key] || 0) - (a[key] || 0));
  renderResultsTable(model, rows);
}

function renderResultsTable(model, rows) {
  const tbody = document.getElementById('results-body-' + model);
  if (!rows.length) { tbody.innerHTML = 'No results'; return; }
  tbody.innerHTML = rows.map(r => {
    const isDrag = r.net_pnl < 0;
    return `
      ${r.symbol}${isDrag ? ' DRAG' : ''}
      ${r.trades}
      ${fmtRs(r.net_pnl)}
      ${r.win_rate}%
      ${r.avg_ret > 0 ? '+' : ''}${r.avg_ret}%
      ${r.avg_hold}d
      
    `;
  }).join('');
}

let _addSym = '';
function openAddModal(sym) {
  _addSym = sym;
  document.getElementById('add-sym').textContent = sym;
  document.getElementById('addBotModal').classList.add('open');
}
function closeAddModal() {
  document.getElementById('addBotModal').classList.remove('open');
}
async function confirmAddBot() {
  const bot = document.getElementById('add-bot-select').value;
  closeAddModal();
  try {
    const res  = await fetch('/api/universe/add', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({bot, symbol: _addSym}),
    });
    const data = await res.json();
    if (data.status === 'added') {
      toast(`${_addSym} added to ${bot}`, 'success');
      await loadUniverse();
    } else if (data.status === 'already_exists') {
      toast(`${_addSym} already in ${bot} universe`, 'warn');
    } else {
      toast(`Error: ${data.message || data.status}`, 'error');
    }
  } catch(e) {
    toast('Network error — try again', 'error');
  }
}

document.getElementById('addBotModal').addEventListener('click', e => {
  if (e.target === document.getElementById('addBotModal')) closeAddModal();
});

loadUniverse();
